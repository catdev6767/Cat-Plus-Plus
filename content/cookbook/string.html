15. Use uppercase letters

purr capitalize(s)
    sniff len(s) == 0
        give s
    paw first = s.char_at(0).upper()
    paw rest = s.substring(1)
    give first + rest

meow capitalize("hello")   # Hello

17. Normalize white space

paw s = "  hello    world  "
meow s.trim()                     # hello    world
meow replace_all("\\s+", s.trim(), " ")  # hello world

18. Check valid email

purr is_email(s)
    give match("^[\\w.]+@[\\w.]+\\.[a-z]+$", s)

meow is_email("a@b.com")   # yes
meow is_email("invalid")   # no

19. Extract number from string

paw text = "Toi co 3 meo va 5 for"
paw nums = find_all("[0-9]+", text)
meow nums   # [3, 5]

20. Number pad to fixed string

purr pad_num(n, width)
    give pad_left(cast(str, n), width, "0")

meow pad_num(5, 3)   # 005
