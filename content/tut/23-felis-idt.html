IDT (Interrupt Descriptor Table) — bảng 256 entry ánh xạ interrupt to handler.

1. Cấu trúc IDT entry

struct idt_entry {
    uint16_t base_lo;
    uint16_t sel;       /* Code segment selector */
    uint8_t  always0;
    uint8_t  flags;     /* 0x8E = present, ring 0, 32-bit */
    uint16_t base_hi;
} __attribute__((packed));

2. Load IDT

struct idt_ptr idtp;
idtp.limit = sizeof(idt) - 1;
idtp.base = (uint32_t)&idt;
asm volatile("lidt (%0)" : : "r"(&idtp));

3. Remap PIC

Mặc định PIC ánh xạ IRQ0-7 → interrupt 0-7. Ta remap để IRQ0-7 → 32-39 (tránh conflict with CPU exceptions).

outb(0x20, 0x11);   /* ICW1 */
outb(0xA0, 0x11);
outb(0x21, 0x20);   /* ICW2 - master offset 32 */
outb(0xA1, 0x28);   /* slave offset 40 */
outb(0x21, 0x04);   /* ICW3 */
outb(0xA1, 0x02);
outb(0x21, 0x01);   /* ICW4 - 8086 mode */
outb(0xA1, 0x01);

4. Keyboard IRQ (IRQ1)

idt_set_gate(33, (uint32_t)irq1_stub, 0x08, 0x8E);

/* Enable IRQ1 */
outb(0x21, 0xFD);  /* 11111101 — chỉ for IRQ1 */
outb(0xA1, 0xFF);  /* Disable slave PIC */

asm volatile("sti");  /* Enable interrupts */

5. IRQ handler

irq1_stub:
    pusha
    call keyboard_handler
    mov $0x20, %al
    outb %al, $0x20       /* EOI - End Of Interrupt */
    popa
    iret
