Cách in ký tự lên màn hình VGA text mode.

1. VGA text mode

Buffer ở địa chỉ0xB8000. 80 cột × 25 hàng. Mỗi ký tự 2 byte.

volatile uint16_t* vga = (uint16_t*)0xB8000;

2. Cấu trúc 1 ký tự

Bit 0-7:  ASCII character
Bit 8-11: Foreground color
Bit 12-15: Background color

uint16_t entry = c | (color << 8);

3. Bảng màu

MãMàu

1Xanh dương
2Xanh lá
4Đỏ

15Trắng

4. Print ký tự

static size_t row = 0, col = 0;
static uint8_t color = 0x0F;  /* Trắng trên đen */

void vga_putc(char c) {
    if (c == '\n') {
        col = 0;
        if (++row >= 25) scroll();
        return;
    }
    vga[row * 80 + col] = c | (color<< 8);
    if (++col >= 80) {
        col = 0;
        if (++row >= 25) scroll();
    }
}

5. Scroll

void scroll(void) {
    for (int y = 1; y < 25; y++)
        for (int x = 0; x < 80; x++)
            vga[(y-1)*80 + x] = vga[y*80 + x];
    for (int x = 0; x < 80; x++)
        vga[24*80 + x] = ' ' | (color << 8);
    row = 24;
}
