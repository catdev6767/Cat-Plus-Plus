26. Tìm max, min

paw nums = [3, 7, 1, 9, 4]

purr find_max(arr)
    paw m = arr[0]
    groom x of arr
        sniff x > m
            paw m = x
    give m

meow find_max(nums)   # 9

28. Nhóm theo điều kiện

purr partition(arr, pred)
    paw yes = []
    paw no = []
    groom x of arr
        sniff pred(x)
            paw yes = yes + [x]
        swat
            paw no = no + [x]
    give [yes, no]

paw result = partition([1,2,3,4,5], kit(x) => x % 2 == 0)
meow result
# [[2, 4], [1, 3, 5]]

29. Cộng dồn

purr accumulate(arr)
    paw result = []
    paw sum = 0
    groom x of arr
        paw sum = sum + x
        paw result = result + [sum]
    give result

meow accumulate([1, 2, 3, 4])
# [1, 3, 6, 10]

30. Zip hai danh sách

purr zip(a, b)
    paw result = []
    paw i = 0
    knead i < len(a) and i < len(b)
        paw result = result + [[a[i], b[i]]]
        i++
    give result

meow zip([1, 2, 3], ["a", "b", "c"])
# [[1, a], [2, b], [3, c]]
