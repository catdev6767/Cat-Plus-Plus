Cách thêm lệnh mới vào Felis OS shell.

1. Viết hàm xử lý

/* Trong commands.c */
static void cmd_greet(char* arg) {
    char buf[128];
    strcpy(buf, "Hello, ");
    strcat(buf, arg);
    catpp_print_str(buf);
}

2. Đăng ký vào command table

const char* shell_commands[] = {
    "help","clear","info",...,
    "greet",              /* ← thêm */
    0
};

3. Thêm vào shell_execute

if (strcmp(cmdline, "greet") == 0) {
    cmd_greet(arg);
    return 0;
}

4. Test

felis$ greet World
Hello, World

5. Examples — lệnh `rand`

static unsigned int seed = 12345;

static void cmd_rand(void) {
    /* Linear congruential generator */
    seed = seed * 1103515245 + 12345;
    unsigned int r = (seed / 65536) % 32768;
    catpp_print(r);
}

/* Trong shell_execute: */
if (strcmp(cmdline, "rand") == 0) { cmd_rand(); return 0; }

Lesson tập
